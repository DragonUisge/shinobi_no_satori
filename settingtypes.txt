# Shinobi no Satori Settings

[Armour]

# Hide shinobi armour from creative inventory.
# When true, armour can only be obtained through the quest.
shinobi_hide_from_creative (Hide armour from creative inventory) bool true

# Heal chance for each armour piece (percentage out of 100).
# With all 3 pieces equipped the total is 3× this value.
shinobi_armour_heal (Armour heal chance per piece) int 18 0 100

[Abilities]

# Damage multiplier for the chestplate.
# 1.0 = normal damage, 2.0 = double damage.
shinobi_damage_multiplier (Chestplate damage multiplier) float 1.8 1.0 5.0

# Movement speed when wall-walking with the headwear.
shinobi_wall_walk_speed (Wall-walk speed) float 4.0 1.0 10.0

# Night vision brightness (0.0 = pitch black, 1.0 = full daylight).
shinobi_night_vision_ratio (Night vision brightness) float 0.6 0.0 1.0

# Speed boost from hakama (added to base speed).
# 0.6 = +60% speed.
shinobi_speed_boost (Hakama speed boost) float 0.6 0.0 3.0

[Shuriken]

# Flight speed of the shuriken (blocks/s).
shinobi_shuriken_speed (Shuriken speed) int 22 5 60

# Maximum range before the shuriken turns back (blocks).
shinobi_shuriken_range (Shuriken range) int 18 5 50

# Punch damage dealt to entities on hit.
shinobi_shuriken_damage (Shuriken damage) int 6 1 20

# Duration of the freeze effect (seconds).
shinobi_shuriken_freeze_time (Freeze duration) float 5.0 0.5 15.0

# Duration of the burn effect (seconds).
shinobi_shuriken_burn_time (Burn duration) float 4.0 0.5 15.0

# Hit detection radius around the shuriken (blocks).
shinobi_shuriken_hit_radius (Hit radius) float 1.8 0.5 5.0

# Cooldown between throws (seconds).
shinobi_shuriken_cooldown (Throw cooldown) float 1.0 0.0 5.0

[Full set bonus]

# Bonus granted when all three armour pieces are worn at the same time.
# Options:
#   nameonly   — the player's nametag is hidden from other players
#   invisible  — the player's body becomes invisible to other players
#   both       — both name and body are hidden
#   scout      — activates a spy ghost: your decoy body stays in place
#                while your shadow goes through the world; press Sneak+Jump to activate and return.
shinobi_set_bonus (Full-set bonus mode) enum nameonly nameonly,invisible,both,scout

# Cooldown between scout activations (seconds).
shinobi_set_bonus_scout_cooldown (Scout cooldown) float 20.0 0.1 120.0
